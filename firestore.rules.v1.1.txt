match /tribunaComments/{commentId} {
  allow create: if
    request.resource.data.keys().hasAll(['displayName','message','status','source','consent','createdAt']) &&
    request.resource.data.keys().hasOnly(['displayName','message','status','source','consent','createdAt']) &&
    request.resource.data.displayName is string &&
    request.resource.data.displayName.size() >= 2 &&
    request.resource.data.displayName.size() <= 40 &&
    request.resource.data.message is string &&
    request.resource.data.message.size() >= 3 &&
    request.resource.data.message.size() <= 300 &&
    request.resource.data.status == 'pending' &&
    request.resource.data.source == 'website' &&
    request.resource.data.consent == true &&
    request.resource.data.createdAt == request.time;

  allow read: if resource.data.status == 'approved' || request.auth != null;
  allow update, delete: if request.auth != null;
}
